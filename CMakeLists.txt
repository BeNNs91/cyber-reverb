cmake_minimum_required(VERSION 3.22)
project(CyberReverb VERSION 0.2.1 LANGUAGES C CXX)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.16-security GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(CyberReverb COMPANY_NAME "CYBER" BUNDLE_ID "com.cyberaudio.cyberreverb"
    PLUGIN_MANUFACTURER_CODE Cybr PLUGIN_CODE CyRv FORMATS VST3 Standalone
    PRODUCT_NAME "cyber-reverb" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
    COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Reverb)
juce_generate_juce_header(CyberReverb)
target_sources(CyberReverb PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_features(CyberReverb PUBLIC cxx_std_17)
target_compile_definitions(CyberReverb PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(CyberReverb PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
enable_testing()
add_executable(ReverbDspTests tests/DspTests.cpp)
target_compile_features(ReverbDspTests PRIVATE cxx_std_17)
target_include_directories(ReverbDspTests PRIVATE Source)
add_test(NAME ReverbDspTests COMMAND ReverbDspTests)
juce_add_console_app(ReverbIntegrationTests PRODUCT_NAME "Reverb Integration Tests")
juce_generate_juce_header(ReverbIntegrationTests)
target_sources(ReverbIntegrationTests PRIVATE tests/IntegrationTests.cpp)
target_include_directories(ReverbIntegrationTests PRIVATE Source)
target_link_libraries(ReverbIntegrationTests PRIVATE CyberReverb juce::juce_audio_utils juce::juce_dsp)
add_test(NAME ReverbIntegrationTests COMMAND ReverbIntegrationTests)
